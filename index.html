// Настройки игры let miniGameType = ""; let miniGameData = []; let miniGameIndex = 0; let miniGameScore = 0; let miniGameCorrectAnswer = ""; // Дефиниции для входов и выходов function initMiniGames(data) { miniGameData = data; miniGameIndex = 0; miniGameScore = 0; document.getElementById('mini-game-screen').style.display = 'block'; document.getElementById('review-screen').style.display = 'none'; } function exitMiniGame() { document.getElementById('mini-game-screen').style.display = 'none'; document.getElementById('review-screen').style.display = 'block'; } function startMiniGame(gameType) { miniGameType = gameType; miniGameIndex = 0; miniGameScore = 0; document.getElementById('mini-game-result').innerText = ""; switch(gameType) { case 'pair': loadPairGame(); break; case 'letter': loadLetterGame(); break; case 'build': loadBuildGame(); break; } } // Игра 1: Найди пару function loadPairGame() { const pairs = []; for (let i = 0; i < miniGameData.length; i++) { const q = miniGameData[i]; const correctWord = q.options[q.correctIndex]; let imageHTML = ""; if (typeof q.image === 'string' && q.image.startsWith('http')) { imageHTML = ''; } else { imageHTML = q.image; } pairs.push({ image: imageHTML, word: correctWord }); } const allCards = []; for (let i = 0; i < pairs.length; i++) { allCards.push({ type: 'image', value: pairs[i].image, pairId: i }); allCards.push({ type: 'word', value: pairs[i].word, pairId: i }); } for (let i = allCards.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [allCards[i], allCards[j]] = [allCards[j], allCards[i]]; } const content = document.getElementById('mini-game-content'); content.innerHTML = '
'; const grid = document.getElementById('pair-grid'); let firstCard = null; let secondCard = null; let lockBoard = false; allCards.forEach((card) => { const cardElement = document.createElement('div'); cardElement.className = 'pair-card'; cardElement.dataset.pairId = card.pairId; cardElement.dataset.type = card.type; cardElement.dataset.value = card.value; if (card.type === 'image') { cardElement.innerHTML = card.value; } else { cardElement.innerHTML = '
' + card.value + '
'; } cardElement.onclick = () => { if (lockBoard) return; if (cardElement.classList.contains('matched')) return; cardElement.classList.add('flipped'); if (!firstCard) { firstCard = cardElement; } else if (!secondCard) { secondCard = cardElement; lockBoard = true; if (firstCard.dataset.pairId === secondCard.dataset.pairId) { firstCard.classList.add('matched'); secondCard.classList.add('matched'); miniGameScore++; firstCard = null; secondCard = null; lockBoard = false; if (miniGameScore === pairs.length) { document.getElementById('mini-game-result').innerText = "All pairs found!"; } } else { setTimeout(() => { firstCard.classList.remove('flipped'); secondCard.classList.remove('flipped'); firstCard = null; secondCard = null; lockBoard = false; }, 1000); } } }; grid.appendChild(cardElement); }); } // Игра 3: Вставь букву function loadLetterGame() { const content = document.getElementById('mini-game-content'); content.innerHTML = '
'; const letterGameDiv = document.querySelector('.letter-game'); const q = miniGameData[miniGameIndex]; const correctWord = q.options[q.correctIndex]; const word = correctWord.toLowerCase(); const letters = word.split(''); // Случайно выбираем пропущенную букву const missingIndex = Math.floor(Math.random() * letters.length); const missingLetter = letters[missingIndex]; const wordWithMissing = word.substring(0, missingIndex) + '_' + word.substring(missingIndex + 1); // Показываем картинку let imageHTML = ""; if (typeof q.image === 'string' && q.image.startsWith('http')) { imageHTML = ''; } else { imageHTML = q.image; } letterGameDiv.innerHTML = imageHTML + '
' + wordWithMissing + '
'; // Буквы для выбора (правильная + 2 неправильных) const lettersToChoose = [missingLetter]; while (lettersToChoose.length < 3) { const randomLetter = String.fromCharCode(97 + Math.floor(Math.random() * 26)); if (!lettersToChoose.includes(randomLetter)) { lettersToChoose.push(randomLetter); } } // Перемешиваем for (let i = lettersToChoose.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [lettersToChoose[i], lettersToChoose[j]] = [lettersToChoose[j], lettersToChoose[i]]; } const letterBtns = document.createElement('div'); letterBtns.className = 'letter-btns'; letterGameDiv.appendChild(letterBtns); lettersToChoose.forEach((letter) => { const btn = document.createElement('button'); btn.className = 'letter-btn'; btn.innerText = letter; btn.onclick = () => { if (letter === missingLetter) { btn.classList.add('correct'); document.getElementById('mini-game-result').innerText = "Correct!"; miniGameScore++; setTimeout(() => { miniGameIndex++; if (miniGameIndex >= miniGameData.length) { document.getElementById('mini-game-result').innerText = "All correct!"; } else { loadLetterGame(); } }, 1000); } else { btn.classList.add('wrong'); btn.disabled = true; } }; letterBtns.appendChild(btn); }); } // Игра 2: Собери слово function loadBuildGame() { const content = document.getElementById('mini-game-content'); content.innerHTML = '
'; const buildGameDiv = document.querySelector('.build-word-game'); const q = miniGameData[miniGameIndex]; const correctWord = q.options[q.correctIndex]; const word = correctWord.toLowerCase(); const letters = word.split(''); // Показываем картинку let imageHTML = ""; if (typeof q.image === 'string' && q.image.startsWith('http')) { imageHTML = ''; } else { imageHTML = q.image; } buildGameDiv.innerHTML = imageHTML + '
'; // Буквы для сборки (перемешанные) const shuffledLetters = [...letters]; for (let i = shuffledLetters.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [shuffledLetters[i], shuffledLetters[j]] = [shuffledLetters[j], shuffledLetters[i]]; } const letterBtns = document.createElement('div'); letterBtns.className = 'build-letter-btns'; buildGameDiv.appendChild(letterBtns); const builtWordDiv = document.getElementById('built-word'); const builtLetters = []; shuffledLetters.forEach((letter, index) => { const btn = document.createElement('button'); btn.className = 'build-letter-btn'; btn.dataset.letter = letter; btn.dataset.index = index; btn.innerText = letter; btn.onclick = () => { btn.classList.add('used'); btn.disabled = true; builtLetters.push(letter); builtWordDiv.innerText = builtLetters.join(''); if (builtLetters.join('') === word) { document.getElementById('mini-game-result').innerText = "Correct!"; miniGameScore++; setTimeout(() => { miniGameIndex++; if (miniGameIndex >= miniGameData.length) { document.getElementById('mini-game-result').innerText = "All correct!"; } else { loadBuildGame(); } }, 1000); } }; letterBtns.appendChild(btn); }); }
