// Общие переменные для игр let miniGameIndex = 0; let miniGameScore = 0; let miniGameCorrectAnswer = ""; let currentGameData = []; function startMiniGame(gameType) { // Получаем данные из index.html currentGameData = window.currentLevelQuestions || []; miniGameIndex = 0; miniGameScore = 0; document.getElementById('mini-game-screen').style.display = 'block'; document.getElementById('review-screen').style.display = 'none'; document.getElementById('mini-game-result').innerText = ""; if (gameType === 'pair') { loadPairGame(); } else if (gameType === 'letter') { loadLetterGame(); } else if (gameType === 'build') { loadBuildGame(); } } function exitMiniGame() { document.getElementById('mini-game-screen').style.display = 'none'; document.getElementById('review-screen').style.display = 'block'; } // Игра 1: Найди пару function loadPairGame() { const grid = document.getElementById('mini-game-content'); grid.innerHTML = '
'; const pairGrid = document.getElementById('pair-grid'); // Создаём пары const pairs = []; for (let i = 0; i < currentGameData.length; i++) { const q = currentGameData[i]; const correctWord = q.options[q.correctIndex]; let image = ""; if (typeof q.image === 'string' && q.image.startsWith('http')) { image = ''; } else { image = q.image; } pairs.push({ image: image, word: correctWord }); } // Перемешиваем const allCards = []; for (let i = 0; i < pairs.length; i++) { allCards.push({ type: 'image', value: pairs[i].image, pairId: i }); allCards.push({ type: 'word', value: pairs[i].word, pairId: i }); } for (let i = allCards.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [allCards[i], allCards[j]] = [allCards[j], allCards[i]]; } let firstCard = null; let secondCard = null; let lockBoard = false; allCards.forEach((card) => { const cardElement = document.createElement('div'); cardElement.className = 'pair-card'; cardElement.dataset.pairId = card.pairId; cardElement.dataset.type = card.type; cardElement.dataset.value = card.value; if (card.type === 'image') { cardElement.innerHTML = card.value; } else { cardElement.innerHTML = '
' + card.value + '
'; } cardElement.onclick = () => { if (lockBoard) return; if (cardElement.classList.contains('matched')) return; cardElement.classList.add('flipped'); if (!firstCard) { firstCard = cardElement; } else if (!secondCard) { secondCard = cardElement; lockBoard = true; if (firstCard.dataset.pairId === secondCard.dataset.pairId) { firstCard.classList.add('matched'); secondCard.classList.add('matched'); firstCard = null; secondCard = null; lockBoard = false; } else { setTimeout(() => { firstCard.classList.remove('flipped'); secondCard.classList.remove('flipped'); firstCard = null; secondCard = null; lockBoard = false; }, 1000); } } }; pairGrid.appendChild(cardElement); }); } // Игра 3: Вставь букву function loadLetterGame() { const content = document.getElementById('mini-game-content'); content.innerHTML = '
'; const letterGame = document.querySelector('.letter-game'); const q = currentGameData[miniGameIndex]; const correctWord = q.options[q.correctIndex]; const word = correctWord.toLowerCase(); const letters = word.split(''); // Случайно выбранная пропущенная буква const missingIndex = Math.floor(Math.random() * letters.length); const missingLetter = letters[missingIndex]; const wordWithMissing = word.substring(0, missingIndex) + '_' + word.substring(missingIndex + 1); let image = ""; if (typeof q.image === 'string' && q.image.startsWith('http')) { image = ''; } else { image = q.image; } letterGame.innerHTML = image + '
' + wordWithMissing + '
'; // Буквы для выборки const lettersToChoose = [missingLetter]; while (lettersToChoose.length < 3) { const randomLetter = String.fromCharCode(97 + Math.floor(Math.random() * 26)); if (!lettersToChoose.includes(randomLetter)) { lettersToChoose.push(randomLetter); } } for (let i = lettersToChoose.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [lettersToChoose[i], lettersToChoose[j]] = [lettersToChoose[j], lettersToChoose[i]]; } const letterBtns = document.createElement('div'); letterBtns.className = 'letter-btns'; letterGame.appendChild(letterBtns); lettersToChoose.forEach((letter) => { const btn = document.createElement('button'); btn.className = 'letter-btn'; btn.innerText = letter; btn.onclick = () => { if (letter === missingLetter) { btn.classList.add('correct'); document.getElementById('mini-game-result').innerText = "Correct!"; miniGameScore++; setTimeout(() => { miniGameIndex++; if (miniGameIndex >= currentGameData.length) { document.getElementById('mini-game-result').innerText = "All correct!"; } else { loadLetterGame(); } }, 1000); } else { btn.classList.add('wrong'); btn.disabled = true; } }; letterBtns.appendChild(btn); }); } // Игра 2: Собери слово function loadBuildGame() { const content = document.getElementById('mini-game-content'); content.innerHTML = '
'; const buildGame = document.querySelector('.build-word-game'); const q = currentGameData[miniGameIndex]; const correctWord = q.options[q.correctIndex]; const word = correctWord.toLowerCase(); const letters = word.split(''); let image = ""; if (typeof q.image === 'string' && q.image.startsWith('http')) { image = ''; } else { image = q.image; } buildGame.innerHTML = image + '
'; // Буквы для сборки const shuffledLetters = [...letters]; for (let i = shuffledLetters.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [shuffledLetters[i], shuffledLetters[j]] = [shuffledLetters[j], shuffledLetters[i]]; } const letterBtns = document.createElement('div'); letterBtns.className = 'build-letter-btns'; buildGame.appendChild(letterBtns); const builtWordDiv = document.getElementById('built-word'); const builtLetters = []; shuffledLetters.forEach((letter, index) => { const btn = document.createElement('button'); btn.className = 'build-letter-btn'; btn.dataset.letter = letter; btn.dataset.index = index; btn.innerText = letter; btn.onclick = () => { btn.classList.add('used'); btn.disabled = true; builtLetters.push(letter); builtWordDiv.innerText = builtLetters.join(''); if (builtLetters.join('') === word) { document.getElementById('mini-game-result').innerText = "Correct!"; miniGameScore++; setTimeout(() => { miniGameIndex++; if (miniGameIndex >= currentGameData.length) { document.getElementById('mini-game-result').innerText = "All correct!"; } else { loadBuildGame(); } }, 1000); } }; letterBtns.appendChild(btn); }); }
